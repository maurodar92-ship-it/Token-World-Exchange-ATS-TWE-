// ── Backend logic ported from FastAPI (runs client-side) ──────────────────────
function seededRandom(seed) {
  let s = seed;
  return () => {
    s = (s * 1664525 + 1013904223) & 0xffffffff;
    return (s >>> 0) / 0xffffffff;
  };
}

function processMatrixPrediction({ resolution, horizon, probability_threshold }) {
  const seed = Math.trunc(resolution * 1000) + horizon * 77;
  const rng = seededRandom(seed);
  const pool = Array.from({ length: 70 }, (_, i) => i + 1);
  const weights = pool.map((_, i) => 1.0 + 0.1 * Math.sin(i * resolution));
  const totalW = weights.reduce((a, b) => a + b, 0);

  const weightedChoice = (excluded) => {
    let r = rng() * totalW;
    for (let i = 0; i < pool.length; i++) {
      if (excluded.includes(pool[i])) continue;
      r -= weights[i];
      if (r <= 0) return pool[i];
    }
    return pool.find((n) => !excluded.includes(n));
  };

  const mainNums = [];
  while (mainNums.length < 5) {
    const c = weightedChoice(mainNums);
    if (c !== undefined) mainNums.push(c);
  }
  mainNums.sort((a, b) => a - b);
  const megaBall = Math.trunc(rng() * 25) + 1;

  const varianceCoef = Math.abs(Math.cos(horizon) * (1.0 - probability_threshold));
  const confidence = Math.min(0.985, Math.max(0.12, probability_threshold - varianceCoef * 0.3));
  const resonance = 432.12 + resolution * 4.67 - horizon * 1.2;
  const quantumDrift = (0.0051 * Math.sin(resolution)) / horizon;

  return {
    status: "TEMPORAL_LOCK_ACHIEVED",
    prediction: { main_numbers: mainNums, mega_ball: megaBall },
    metrics: {
      confidence_rating: `${(confidence * 100).toFixed(2)}%`,
      resonance_frequency: `${resonance.toFixed(2)} THz`,
      quantum_drift: `${quantumDrift.toFixed(6)}s`,
    },
  };
}

// ── Pulsing ring animation around mega ball ───────────────────────────────────
function PulseRing() {
  return (
    <span className="absolute inset-0 rounded-full animate-ping bg-red-500 opacity-20" />
  );
}

// ── Number Ball component ──────────────────────────────────────────────────────
function Ball({ n, mega = false, delay = 0 }) {
  const [visible, setVisible] = useState(false);
  useEffect(() => {
    const t = setTimeout(() => setVisible(true), delay);
    return () => clearTimeout(t);
  }, [delay]);

  return (
    <div
      className={`relative flex items-center justify-center font-black text-lg rounded-full transition-all duration-500 ${
        visible ? "opacity-100 scale-100" : "opacity-0 scale-50"
      } ${
        mega
          ? "w-14 h-14 bg-red-900 border-2 border-red-400 text-red-200 shadow-lg shadow-red-500/40"
          : "w-12 h-12 bg-cyan-950 border-2 border-cyan-400 text-cyan-100 shadow-md shadow-cyan-400/30"
      }`}
      style={{ fontVariantNumeric: "tabular-nums" }}
    >
      {mega && <PulseRing />}
      <span className="relative z-10">{n}</span>
    </div>
  );
}

// ── Animated metric counter ────────────────────────────────────────────────────
function MetricCard({ label, value, accent = "cyan" }) {
  const colors = {
    cyan: "border-cyan-800 text-cyan-300",
    amber: "border-amber-800 text-amber-300",
    green: "border-green-800 text-green-300",
    purple: "border-purple-800 text-purple-300",
  };
  return (
    <div className={`border ${colors[accent]} bg-slate-900/70 rounded p-3`}>
      <div className="text-[10px] tracking-widest text-slate-500 uppercase mb-1">{label}</div>
      <div className={`text-base font-mono font-bold ${colors[accent].split(" ")[1]}`}>{value}</div>
    </div>
  );
}

// ── Sweep progress bar ─────────────────────────────────────────────────────────
function SweepBar({ active }) {
  return (

      <div
        className={`h-full bg-cyan-400 transition-all duration-[2200ms] ease-in-out ${

// ── Slider row ─────────────────────────────────────────────────────────────────
function SliderRow({ label, value, min, max, step = 1, unit, onChange }) {
  return (
    <div className="space-y-1">
      <div className="flex justify-between text-[11px] tracking-widest">
        <span className="text-cyan-700 uppercase">{label}</span>
        <span className="text-white font-bold tabular-nums">
          {typeof value === "number" && step < 1
            ? `${(value * 100).toFixed(0)}`
            : value}{" "}
          <span className="text-cyan-600">{unit}</span>
        </span>
      </div>
      <div className="relative">
        <input
          type="range"
          min={min}
          max={max}
          step={step}
          value={value}
          onChange={(e) =>
            onChange(step < 1 ? parseFloat(e.target.value) : parseInt(e.target.value))
          }
          className="w-full h-2 appearance-none bg-slate-800 border border-cyan-900 rounded cursor-pointer"
          style={{
            accentColor: "#22d3ee",
          }}
        />
        <div
          className="absolute top-0 left-0 h-2 bg-cyan-500/30 rounded pointer-events-none"
          style={{ width: `${((value - min) / (max - min)) * 100}%` }}
        />
      </div>
    </div>
  );
}

// ── Blinking cursor ────────────────────────────────────────────────────────────
function Cursor() {
  const [on, setOn] = useState(true);
  useEffect(() => {
    const t = setInterval(() => setOn((p) => !p), 530);
    return () => clearInterval(t);
  }, []);
  return <span className={on ? "opacity-100" : "opacity-0"}>█</span>;
}

// ── Log terminal ───────────────────────────────────────────────────────────────
function Terminal({ lines }) {
  const ref = useRef(null);
  useEffect(() => {
    if (ref.current) ref.current.scrollTop = ref.current.scrollHeight;
  }, [lines]);
  return (
    <div
      ref={ref}
      className="bg-black border border-cyan-900 rounded p-3 h-36 overflow-y-auto text-[11px] font-mono space-y-0.5"
    >
      {lines.map((l, i) => (
        <div key={i} className={l.color || "text-cyan-700"}>
          {l.text}
        </div>
      ))}
      <div className="text-cyan-400">
        &gt; <Cursor />
      </div>
    </div>
  );
}
