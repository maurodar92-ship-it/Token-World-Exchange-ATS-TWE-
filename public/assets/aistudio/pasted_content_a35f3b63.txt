const MiniChart = ({ symbol }) => {
  const canvasRef = useRef(null);
  useEffect(() => {
    const c = canvasRef.current; if (!c) return;
    const ctx = c.getContext("2d");
    const pts = Array.from({ length: 30 }, (_, i) => 50 + Math.sin(i * 0.4) * 15 + Math.random() * 10 - 5 + i * 0.8);
    const min = Math.min(...pts), max = Math.max(...pts);
    const sx = c.width / (pts.length - 1), sy = c.height / (max - min);
    ctx.clearRect(0, 0, c.width, c.height);
    const grad = ctx.createLinearGradient(0, 0, 0, c.height);
    grad.addColorStop(0, "rgba(200,168,75,0.3)");
    grad.addColorStop(1, "rgba(200,168,75,0)");
    ctx.beginPath();
    pts.forEach((p, i) => {
      const x = i * sx, y = c.height - (p - min) * sy;
      i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
    });
    ctx.strokeStyle = "#c8a84b"; ctx.lineWidth = 1.5; ctx.stroke();
    ctx.lineTo(c.width, c.height); ctx.lineTo(0, c.height); ctx.closePath();
    ctx.fillStyle = grad; ctx.fill();
  }, []);
  return <canvas ref={canvasRef} width={80} height={28} style={{ display: "block" }} />;
};
