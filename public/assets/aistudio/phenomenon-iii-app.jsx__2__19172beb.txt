// ---------- Ambient background: starfield ----------
function StarField() {
  const stars = Array.from({ length: 100 }, (_, i) => ({
    id: i,
    x: Math.random() * 100,
    y: Math.random() * 100,
    size: Math.random() * 2 + 0.5,
    delay: Math.random() * 5,
    color: i % 7 === 0 ? COLOR.cyan : i % 11 === 0 ? COLOR.pink : "#ffffff",
  }));

  return (
    <div className="fixed inset-0 cosmic-bg pointer-events-none" style={{ zIndex: 0 }}>
      {stars.map((star) => (
        <div
          key={star.id}
          className="absolute rounded-full animate-pulse-glow"
          style={{
            left: `${star.x}%`,
            top: `${star.y}%`,
            width: star.size,
            height: star.size,
            background: star.color,
            boxShadow: `0 0 ${star.size * 2}px ${star.color}`,
            animationDelay: `${star.delay}s`,
          }}
        />
      ))}
      <div className="absolute inset-0 bg-gradient-to-b from-transparent via-transparent to-black/40" />
    </div>
  );
}

// ---------- Hexagram seal (interactive) ----------
function HexSeal({ interactive = false, onClick, active, size = 220 }) {
  const c = size / 2;
  const rInner = size * 0.35;
  const rStar = size * 0.48;

  const triUp = Array.from({ length: 3 }, (_, i) => {
    const a = -Math.PI / 2 + (i * 2 * Math.PI) / 3;
    return `${c + rInner * Math.cos(a)},${c + rInner * Math.sin(a)}`;
  }).join(" ");

  const triDown = Array.from({ length: 3 }, (_, i) => {
    const a = Math.PI / 2 + (i * 2 * Math.PI) / 3;
    return `${c + rInner * Math.cos(a)},${c + rInner * Math.sin(a)}`;
  }).join(" ");

  const starPoints = Array.from({ length: 16 }, (_, i) => {
    const a = (i * 2 * Math.PI) / 16 - Math.PI / 2;
    return { x: c + rStar * Math.cos(a), y: c + rStar * Math.sin(a), id: i };
  });

  const rayBase = size * 0.4;

  return (
    <svg
      viewBox={`0 0 ${size} ${size}`}
      className={interactive ? "cursor-pointer" : ""}
      style={{ width: size, height: size }}
      onClick={onClick}
    >
      <circle cx={c} cy={c} r={rStar + 8} fill="none" stroke={active ? COLOR.cyan : "#1e3a5f"}
        strokeWidth="1.5" strokeDasharray="4 3" className="animate-rotate-slow" style={{ transformOrigin: `${c}px ${c}px` }} />

      {starPoints.map((p) => (
        <text key={p.id} x={p.x} y={p.y} textAnchor="middle" dominantBaseline="central"
          fontSize="8" fill={active ? COLOR.gold : COLOR.dim}>★</text>
      ))}

      <circle cx={c} cy={c} r={rStar} fill={active ? `${COLOR.cyan}08` : "none"}
        stroke={active ? COLOR.cyan : COLOR.dim} strokeWidth="1" />

      {[80, 95, 110, 125, 140, 155, 170].map((deg) => {
        const rad = (deg * Math.PI) / 180;
        const inner = rayBase * 0.3, outer = rayBase * 0.85;
        return (
          <line key={deg} x1={c + inner * Math.cos(rad)} y1={c + inner * Math.sin(rad)}
            x2={c + outer * Math.cos(rad)} y2={c + outer * Math.sin(rad)}
            stroke={active ? COLOR.gold : COLOR.dim} strokeWidth="0.5" opacity="0.5" />
        );
      })}

      {[30, 90, 150, 210, 270, 330].map((deg) => {
        const rad = (deg * Math.PI) / 180;
        return (
          <circle key={deg} cx={c + (rayBase + 12) * Math.cos(rad)} cy={c + (rayBase + 12) * Math.sin(rad)}
            r="5" fill="none" stroke={active ? COLOR.cyan : COLOR.dim} strokeWidth="1.5" />
        );
      })}

      <polygon points={triUp} fill="none" stroke={active ? COLOR.gold : "#fff"} strokeWidth="2.5" strokeLinejoin="round" />
      <polygon points={triDown} fill="none" stroke={active ? COLOR.gold : "#fff"} strokeWidth="2.5" strokeLinejoin="round" />

      {Array.from({ length: 24 }, (_, i) => {
        const a = (i * Math.PI * 2) / 24;
        const inner = rInner * 0.3, outer = rInner * 0.65;
        return (
          <line key={`ray-${i}`} x1={c + inner * Math.cos(a)} y1={c + inner * Math.sin(a)}
            x2={c + outer * Math.cos(a)} y2={c + outer * Math.sin(a)}
            stroke={active ? COLOR.gold : COLOR.cyan} strokeWidth="0.8" opacity="0.7" />
        );
      })}

      <circle cx={c} cy={c} r="7" fill={active ? COLOR.gold : "#fff"} />
    </svg>
  );
}
